# Create a clean dump with structure and all source/config
(
  echo "=== PROJECT STRUCTURE ==="
  find . -path "*/\.git" -prune -o -type d -print | sort | sed -e 's/[^-][^\/]*\//  |/g' -e 's/|\([^ ]\)/|-\1/' 
  echo -e "\n\n=== SOURCE FILES ===\n"
  find . -type f \
    \( -name "*.java" -o \
       -name "pom.xml" -o \
       -name "*.md" -o \
       -name "application*.properties" -o \
       -name "*.yml" -o \
       -name "*.yaml" -o \
       -name "Dockerfile" -o \
       -name "*.sql" \) \
    ! -path "*/\.git/*" \
    ! -path "*/target/*" \
    ! -path "*/\.idea/*" \
    -exec echo -e "\n\n--- {} ---\n" \; \
    -exec cat {} \;
) > current_state1.txt
